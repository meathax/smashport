cmake_minimum_required(VERSION 3.20)
project(SmashTVNative VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Only Windows x64 is supported")
endif()
option(STV_PLAYER_RELEASE "Build fixed shared 16:9 player host; reference geometry remains in development tools" OFF)
option(STV_SANITIZE "Enable ASan and UBSan on host tools/tests" OFF)
option(STV_LAB "Also build the retired v0.5 hand-written lab code (SmashTVLab, stv_core) and its tests" OFF)
# A default configure must never succeed with only utility tests or without audio.
# These private outputs are intentionally absent from Git: generate them locally.
set(STV_PLAYER_INPUTS "" CACHE PATH "Native sources generated from the selected ROM")
set(STV_SDL_SOURCE "" CACHE PATH "Optional local SDL3 source checkout")
if(STV_PLAYER_INPUTS)
 add_subdirectory(third_party/recomp_achievements EXCLUDE_FROM_ALL)
 if(STV_SDL_SOURCE)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  add_subdirectory("${STV_SDL_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/sdl" EXCLUDE_FROM_ALL)
 else()
  find_package(SDL3 3.2 CONFIG QUIET)
 endif()
 if(NOT WIN32 AND NOT TARGET SDL3::SDL3)
  message(FATAL_ERROR "The selected-ROM player needs SDL3 3.2 or newer. Set SDL3_DIR or STV_SDL_SOURCE.")
 endif()
endif()
set(STV_GENERATED "${CMAKE_CURRENT_SOURCE_DIR}/generated/private")
if(STV_PLAYER_INPUTS)
 if(NOT EXISTS "${STV_PLAYER_INPUTS}/outputs.json")
  message(FATAL_ERROR "Player sources must be prepared from the selected retail L8 ZIP by build.ps1.")
 endif()
 set(STV_GENERATED "${STV_PLAYER_INPUTS}")
endif()
foreach(STV_REQUIRED_GENERATED IN ITEMS l8/l8_dispatch.cpp l8/l8_chunk_000.cpp l8/enhanced_stage_balance.inc snd/snd_code.cpp)
 if(NOT EXISTS "${STV_GENERATED}/${STV_REQUIRED_GENERATED}")
  message(FATAL_ERROR "Missing generated/private/${STV_REQUIRED_GENERATED}. From the project root run: python tools/gsp2cpp/codemap.py, python tools/gsp2cpp/ensure_emission.py --refresh, and python tools/m6809_aot/emit6809.py original/user-supplied-roms/smashtv.zip. See README.md for source-alignment prerequisites.")
 endif()
endforeach()
# Shared host support used by the translated game: base types and the ROM ZIP reader.
add_library(stv_base STATIC src/base.cpp src/zip.cpp)
target_include_directories(stv_base PUBLIC include)
if(MSVC)
 target_compile_options(stv_base PRIVATE /W4 /permissive-)
else()
 target_compile_options(stv_base PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(STV_SANITIZE AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
# PC presentation and room UI, adapted from TMNT-port/arcade-lan. This library
# owns no translated machine, simulation clock, snapshots or confirmed audio.
add_library(stv_pc_frontend STATIC src/host/pc_menu.cpp src/host/pc_settings.cpp src/host/variant_toast.cpp src/host/video_filter.cpp src/host/enhance.cpp src/host/gpu_effects.cpp src/host/ray_scene.cpp src/host/room_lighting.cpp src/host/audio_fx.cpp src/net/lobby.cpp)
target_include_directories(stv_pc_frontend PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(stv_pc_frontend PRIVATE STV_PLAYER_RELEASE=$<BOOL:${STV_PLAYER_RELEASE}>)
if(WIN32)
 # Global online leaderboards (shared recomp client; server: Cloudflare Worker + D1).
 # STV_LEADERBOARD_URL in the environment overrides the URL at run time.
 set(STV_LEADERBOARD_URL "https://recomp-leaderboard.meathax.workers.dev" CACHE STRING "Recomp leaderboard Worker URL")
 add_subdirectory(third_party/recomp_leaderboard EXCLUDE_FROM_ALL)
 target_link_libraries(stv_pc_frontend PUBLIC rl_leaderboard)
 target_compile_definitions(stv_pc_frontend PUBLIC "STV_LEADERBOARD_URL=L\"${STV_LEADERBOARD_URL}\"")
 set(STV_EOS_SDK "${CMAKE_CURRENT_SOURCE_DIR}/build/dependencies/eos/1.19.2.1/SDK" CACHE PATH "Epic Online Services C SDK")
 if(EXISTS "${STV_EOS_SDK}/Include/eos_sdk.h")
  target_sources(stv_pc_frontend PRIVATE src/net/epic_rooms.cpp)
  target_include_directories(stv_pc_frontend PRIVATE "${STV_EOS_SDK}/Include")
  target_link_libraries(stv_pc_frontend PRIVATE "${STV_EOS_SDK}/Lib/EOSSDK-Win64-Shipping.lib")
  target_compile_definitions(stv_pc_frontend PUBLIC STV_HAS_EOS=1)
  set(STV_EOS_CONFIG "${CMAKE_CURRENT_SOURCE_DIR}/build/dependencies/eos/eos-client.ini" CACHE FILEPATH "Local EOS game client configuration")
  if(STV_PLAYER_RELEASE OR EXISTS "${STV_EOS_CONFIG}")
   configure_file("${STV_EOS_CONFIG}" "${CMAKE_CURRENT_BINARY_DIR}/eos-client.ini" COPYONLY)
  endif()
  configure_file("${STV_EOS_SDK}/Bin/EOSSDK-Win64-Shipping.dll" "${CMAKE_CURRENT_BINARY_DIR}/EOSSDK-Win64-Shipping.dll" COPYONLY)
 elseif(STV_PLAYER_RELEASE)
  message(FATAL_ERROR "Windows player builds require the EOS SDK for public multiplayer")
 endif()
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/host/resident_effects.hlsl")
 file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/host/resident_effects.hlsl" STV_EFFECTS_HLSL)
 configure_file(src/host/resident_effects_shader.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/resident_effects_shader.hpp" @ONLY)
 target_include_directories(stv_pc_frontend PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
 # Compile DXIL once during the Windows build. The player needs only the driver.
 find_program(STV_DXC dxc HINTS "$ENV{WindowsSdkVerBinPath}/x64"
              "$ENV{WindowsSdkDir}/bin/$ENV{WindowsSDKVersion}/x64" REQUIRED)
 get_filename_component(STV_DXC_BIN "${STV_DXC}" DIRECTORY)
 find_program(STV_FXC fxc HINTS "${STV_DXC_BIN}" "$ENV{WindowsSdkVerBinPath}/x64" REQUIRED)
 find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
 set(STV_SHADER_BUILDER "${CMAKE_CURRENT_SOURCE_DIR}/tools/release/installer/compile_shaders.py")
 set(STV_SHADER_HEADER "${CMAKE_CURRENT_BINARY_DIR}/resident_bytecode.hpp")
 add_custom_command(OUTPUT "${STV_SHADER_HEADER}"
   COMMAND "${Python3_EXECUTABLE}" "${STV_SHADER_BUILDER}" "${STV_FXC}" "${STV_SHADER_HEADER}"
   DEPENDS "${STV_SHADER_BUILDER}" src/host/resident_effects.hlsl src/host/resident_display.hlsl src/host/resident_menu.hlsl
   VERBATIM)
 add_custom_target(stv_resident_bytecode DEPENDS "${STV_SHADER_HEADER}")
 add_dependencies(stv_pc_frontend stv_resident_bytecode)
 set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/host/dxr_lighting.hlsl")
 execute_process(
   COMMAND "${STV_DXC}" -T cs_6_5 -E main -O3 -Fh "${CMAKE_CURRENT_BINARY_DIR}/dxr_shader.hpp.new"
           -Vn stv_dxr_shader "${CMAKE_CURRENT_SOURCE_DIR}/src/host/dxr_lighting.hlsl"
   COMMAND_ERROR_IS_FATAL ANY)
 # Replace the header only when the DXIL changed: rewriting it on every
 # configure recompiled dxr_lighting.cpp and relinked every executable.
 configure_file("${CMAKE_CURRENT_BINARY_DIR}/dxr_shader.hpp.new" "${CMAKE_CURRENT_BINARY_DIR}/dxr_shader.hpp" COPYONLY)
 target_sources(stv_pc_frontend PRIVATE src/host/dxr_lighting.cpp src/host/gpu_resident.cpp src/host/gpu_filter.cpp third_party/graphics/runtime/post/post_chain.cpp)
 target_include_directories(stv_pc_frontend PRIVATE third_party/graphics/runtime/post third_party/graphics/third_party/librashader/include)
 target_compile_definitions(stv_pc_frontend PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(stv_pc_frontend PRIVATE ws2_32 d3d11 d3d12 dxgi d3dcompiler)
 configure_file(third_party/graphics/librashader-d3d11/librashader_d3d11.dll ${CMAKE_CURRENT_BINARY_DIR}/librashader_d3d11.dll COPYONLY)
 configure_file(third_party/graphics/third_party/librashader/bin/librashader.dll ${CMAKE_CURRENT_BINARY_DIR}/librashader.dll COPYONLY)
 configure_file(third_party/graphics/third_party/librashader/dxc/dxcompiler.dll ${CMAKE_CURRENT_BINARY_DIR}/dxcompiler.dll COPYONLY)
 file(COPY third_party/graphics/shaders DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
endif()
if(MSVC)
 target_compile_options(stv_pc_frontend PRIVATE /W4 /permissive-)
else()
 target_compile_options(stv_pc_frontend PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(stv_gsp STATIC src/gsp/machine.cpp src/gsp/gfx.cpp src/gsp/run.cpp src/gsp/enhanced.cpp src/gsp/state.cpp src/gsp/variants.cpp)
target_include_directories(stv_gsp PUBLIC include PRIVATE "${STV_GENERATED}/l8")
target_link_libraries(stv_gsp PUBLIC stv_base)
# ---- Translated sound program (PLAN P6): MC6809E code from the user's sound ROMs,
# python tools/m6809_aot/emit6809.py; YM2151 through vendored ymfm (MAME 0.289, BSD-3).
set(STV_SND_GENERATED ${STV_GENERATED}/snd/snd_code.cpp)
if(EXISTS ${STV_SND_GENERATED})
 add_library(stv_snd STATIC src/snd/board.cpp src/snd/audio.cpp src/snd/voices.cpp third_party/ymfm/ymfm_opm.cpp ${STV_SND_GENERATED})
 target_include_directories(stv_snd PUBLIC include PRIVATE third_party/ymfm)
 if(MSVC)
  set_source_files_properties(${STV_SND_GENERATED} third_party/ymfm/ymfm_opm.cpp PROPERTIES COMPILE_OPTIONS "/bigobj;/W0")
 else()
  set_source_files_properties(${STV_SND_GENERATED} third_party/ymfm/ymfm_opm.cpp PROPERTIES COMPILE_OPTIONS "-w")
 endif()
endif()
file(GLOB STV_L8_GENERATED CONFIGURE_DEPENDS ${STV_GENERATED}/l8/l8_*.cpp)
if(STV_L8_GENERATED)
 add_library(stv_l8 STATIC ${STV_L8_GENERATED})
 find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
 if(NOT STV_PLAYER_INPUTS)
 add_custom_target(stv_l8_emission_check
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/gsp2cpp/ensure_emission.py"
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
  COMMENT "Checking private L8 emission freshness"
  VERBATIM)
 add_dependencies(stv_l8 stv_l8_emission_check)
 endif()
 target_link_libraries(stv_l8 PUBLIC stv_gsp)
 if(MSVC)
  target_compile_options(stv_l8 PRIVATE /bigobj /W0)
 else()
  target_compile_options(stv_l8 PRIVATE -w)
 endif()
 if(WIN32 AND STV_PLAYER_INPUTS)
  # SmashTV.exe: the translated L8 game on the native Windows host
  enable_language(RC)
  add_executable(SmashTV WIN32 src/host/win_main.cpp src/host/wasapi.cpp src/host/resources/smash.rc)
  target_include_directories(SmashTV PRIVATE src/host/resources)
  target_compile_definitions(SmashTV PRIVATE UNICODE _UNICODE)
  target_compile_definitions(SmashTV PRIVATE STV_PLAYER_RELEASE=$<BOOL:${STV_PLAYER_RELEASE}>)
  target_compile_definitions(SmashTV PRIVATE STV_RECOMP_ACHIEVEMENTS)
  target_link_libraries(SmashTV PRIVATE rl_achievements)
  if(TARGET SDL3::SDL3)
   target_compile_definitions(SmashTV PRIVATE STV_SDL_PADS)
   target_link_libraries(SmashTV PRIVATE SDL3::SDL3)
   if(TARGET SDL3::SDL3-shared)
    add_custom_command(TARGET SmashTV POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
     $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:SmashTV> VERBATIM)
   endif()
  endif()
  # stv_game (defined below) = translated main + sound programs under the machine scheduler
  target_link_libraries(SmashTV PRIVATE stv_game stv_net stv_pc_frontend kernel32 user32 gdi32 comdlg32 ole32 shell32)
  if(MINGW)
   target_compile_definitions(SmashTV PRIVATE WINVER=0x0a00 _WIN32_WINNT=0x0a00)
   target_link_libraries(SmashTV PRIVATE ksguid)
   target_link_options(SmashTV PRIVATE -municode)
  endif()
  if(MSVC)
   target_compile_options(SmashTV PRIVATE /W4 /permissive-)
   target_link_options(SmashTV PRIVATE /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT)
  endif()
 endif()
 if(NOT WIN32 AND STV_PLAYER_INPUTS)
  add_executable(SmashTV src/host/sdl_main.cpp)
  target_compile_definitions(SmashTV PRIVATE STV_RECOMP_ACHIEVEMENTS)
  # Unix archive linkers resolve left to right. The GSP run loop asks the
  # translated archive for the chunk to dispatch, so scan that archive next.
  target_link_libraries(SmashTV PRIVATE stv_game stv_gsp stv_l8 stv_pc_frontend SDL3::SDL3 rl_achievements)
  if(APPLE)
   set_target_properties(SmashTV PROPERTIES MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_BUNDLE_NAME "Smash T.V."
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.smashport.player")
  endif()
 endif()
  add_library(stv_game STATIC src/game/cosched.cpp)
  target_link_libraries(stv_game PUBLIC stv_l8 stv_snd)
 add_library(stv_net STATIC src/net/rollback.cpp src/net/handshake.cpp src/net/udp.cpp)
 target_link_libraries(stv_net PUBLIC stv_l8)
 if(WIN32)
  target_link_libraries(stv_net PUBLIC ws2_32)
 endif()
endif()
